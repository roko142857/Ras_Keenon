cmake_minimum_required(VERSION 3.8)
project(ras_keenon_native)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
set(QL_SDK_ROOT "/home/peanut/homebody/vendor/ql_sdk" CACHE PATH "Matched ARM64 SDK directory")
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  message(FATAL_ERROR "Native SDK is ARM64. Build this package on the robot.")
endif()
foreach(FILE include/qlrobotarm_canfd.h lib/libqlrobot_control.a lib/libcanFD2.so)
  if(NOT EXISTS "${QL_SDK_ROOT}/${FILE}")
    message(FATAL_ERROR "Missing ${QL_SDK_ROOT}/${FILE}")
  endif()
endforeach()
file(SHA256 "${QL_SDK_ROOT}/lib/libqlrobot_control.a" QL_SDK_HASH)
if(NOT QL_SDK_HASH STREQUAL "77cdc6c9e4e37dbd0d3738bdfe7cdca615e8bb751c1c82d5c4cb22db29dfb6b1")
  message(FATAL_ERROR "CAN callback ABI is verified only for the pinned HomeBody SDK archive")
endif()
add_executable(sdk_bridge src/sdk_bridge.cpp)
target_compile_features(sdk_bridge PRIVATE cxx_std_14)
target_include_directories(sdk_bridge PRIVATE "${QL_SDK_ROOT}/include")
target_link_libraries(sdk_bridge PRIVATE Eigen3::Eigen
  "-Wl,--start-group" "${QL_SDK_ROOT}/lib/libqlrobot_control.a" "-Wl,--end-group"
  "${QL_SDK_ROOT}/lib/libcanFD2.so" Threads::Threads m)
target_link_libraries(sdk_bridge PRIVATE "-Wl,--wrap=_Z11hal_receivePFvP15hal_data_recv_tiE")
set_target_properties(sdk_bridge PROPERTIES BUILD_RPATH "${QL_SDK_ROOT}/lib" INSTALL_RPATH "${QL_SDK_ROOT}/lib")
install(TARGETS sdk_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
